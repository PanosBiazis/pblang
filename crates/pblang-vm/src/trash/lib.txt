//TreeVM(lib.rs) Used and necessary for the rust compiler :(

use sapphire_core::{Expr, Stmt};
use std::collections::HashMap;

pub mod chunk;
pub mod vm;


#[derive(Clone, Debug, PartialEq)]
pub enum Value {
    Number(f64),
    Str(String),
    Bool(bool),
    Nil,
}

impl Value {
    pub fn as_str(&self) -> Result<&str, String> {
        match self {
            Value::Str(s) => Ok(s),
            _ => Err("expected string".to_string()),
        }
    }

    pub fn add(self, rhs: Value) -> Result<Value, String> {
        match (self, rhs) {
            (Value::Number(a), Value::Number(b)) => Ok(Value::Number(a + b)),
            (Value::Str(a), Value::Str(b)) => Ok(Value::Str(format!("{a}{b}"))),
            (a, b) => Ok(Value::Str(format!("{a}{b}"))), // matches your current REPL concat behavior
        }
    }
}


fn fmt_f64_default(n: f64) -> String {
    let mut s = format!("{:.10}", n);
    if s.contains('.') {
        while s.ends_with('0') {
            s.pop();
        }
        if s.ends_with('.') {
            s.pop();
        }
    }
    s
}

impl std::fmt::Display for Value {
    fn fmt(&self, f: &mut std::fmt::Formatter) -> std::fmt::Result {
        match self {
            Value::Number(n) => write!(f, "{}", fmt_f64_default(*n)),
            Value::Str(s) => write!(f, "{}", s),
            Value::Bool(b) => write!(f, "{}", b),
            Value::Nil => write!(f, "nil"),
        }
    }
}

pub struct VM {
    pub vars: HashMap<String, Value>,
}

impl VM {
    pub fn new() -> Self {
        VM {
            vars: HashMap::new(),
        }
    }

    // Keep your external API (takes Vec<Stmt>), but internally run by reference.
    pub fn run(&mut self, stmts: Vec<Stmt>) {
        self.run_block(&stmts);
    }

    fn run_block(&mut self, stmts: &[Stmt]) {
        for stmt in stmts {
            match stmt {
                Stmt::Print(expr) => {
                    let val = self.eval(expr);
                    println!("{}", val);
                }

                Stmt::PrintF(p, expr) => match self.eval(expr) {
                    Value::Number(n) => println!("{:.*}", *p, n),
                    v => println!("{}", v),
                },

                Stmt::Let(name, expr) => {
                    let value = self.eval(expr);
                    self.vars.insert(name.clone(), value);
                }

                Stmt::If(cond, body) => {
                    if let Value::Bool(true) = self.eval(cond) {
                        self.run_block(body);
                    }
                }

                Stmt::While(cond, body) => {
                    while matches!(self.eval(cond), Value::Bool(true)) {
                        self.run_block(body);
                    }
                }

                // for i N ( ... )  => repeat N times, i = 0..N-1
                Stmt::For(name, count_expr, body) => {
                    let n = match self.eval(count_expr) {
                        Value::Number(x) if x.is_finite() => x as i64,
                        _ => 0,
                    };

                    let n = n.max(0);

                    for i in 0..n {
                        self.vars.insert(name.clone(), Value::Number(i as f64));
                        self.run_block(body);
                    }
                }
            }
        }
    }

    fn eval(&self, expr: &Expr) -> Value {
        match expr {
            Expr::Number(n) => Value::Number(*n),
            Expr::Str(s) => Value::Str(s.clone()),
            Expr::Bool(b) => Value::Bool(*b),

            Expr::Var(v) => self.vars.get(v).cloned().unwrap_or(Value::Nil),

            Expr::Add(a, b) => match (self.eval(a), self.eval(b)) {
                (Value::Number(l), Value::Number(r)) => Value::Number(l + r),
                (Value::Str(l), Value::Str(r)) => Value::Str(format!("{}{}", l, r)),
                (l, r) => Value::Str(format!("{}{}", l, r)),
            },

            Expr::Sub(a, b) => match (self.eval(a), self.eval(b)) {
                (Value::Number(l), Value::Number(r)) => Value::Number(l - r),
                _ => Value::Nil,
            },

            Expr::Mul(a, b) => match (self.eval(a), self.eval(b)) {
                (Value::Number(l), Value::Number(r)) => Value::Number(l * r),
                _ => Value::Nil,
            },

            Expr::Div(a, b) => match (self.eval(a), self.eval(b)) {
                (Value::Number(_), Value::Number(r)) if r == 0.0 => Value::Nil,
                (Value::Number(l), Value::Number(r)) => Value::Number(l / r),
                _ => Value::Nil,
            },

            Expr::Eq(a, b) => {
                let left = self.eval(a);
                let right = self.eval(b);
                Value::Bool(match (left, right) {
                    (Value::Nil, _) | (_, Value::Nil) => false,
                    (l, r) => l == r,
                })
            }

            Expr::NotEq(a, b) => {
                let left = self.eval(a);
                let right = self.eval(b);
                Value::Bool(match (left, right) {
                    (Value::Nil, _) | (_, Value::Nil) => true,
                    (l, r) => l != r,
                })
            }

            Expr::Gt(a, b) => match (self.eval(a), self.eval(b)) {
                (Value::Number(l), Value::Number(r)) => Value::Bool(l > r),
                _ => Value::Bool(false),
            },

            Expr::Ge(a, b) => match (self.eval(a), self.eval(b)) {
                (Value::Number(l), Value::Number(r)) => Value::Bool(l >= r),
                _ => Value::Bool(false),
            },

            Expr::Lt(a, b) => match (self.eval(a), self.eval(b)) {
                (Value::Number(l), Value::Number(r)) => Value::Bool(l < r),
                _ => Value::Bool(false),
            },

            Expr::Le(a, b) => match (self.eval(a), self.eval(b)) {
                (Value::Number(l), Value::Number(r)) => Value::Bool(l <= r),
                _ => Value::Bool(false),
            },
        }
    }
}
