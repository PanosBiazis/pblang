use crate::chunk::{Chunk, OpCode};
use crate::{Closure, Function, Value};

use std::collections::HashMap;
use std::fmt;
use std::rc::Rc;

impl fmt::Display for Value {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Value::Number(n) => write!(f, "{n}"),
            Value::Str(s) => write!(f, "{s}"),
            Value::Bool(b) => write!(f, "{b}"),
            Value::Nil => write!(f, "nil"),
            Value::Function(_) => write!(f, "fn"),
            Value::Closure(_) => write!(f, "closure"),
        }
    }
}

#[derive(Debug)]
struct CallFrame {
    closure: Rc<Closure>,
    ip: usize,
    stack_base: usize,
}

pub struct VM {
    stack: Vec<Value>,
    globals: HashMap<String, Value>,
    frames: Vec<CallFrame>,
}

impl VM {
    pub fn new() -> Self {
        Self {
            stack: vec![],
            globals: HashMap::new(),
            frames: vec![],
        }
    }

    pub fn run(&mut self, chunk: Chunk) -> Result<(), String> {
        self.stack.clear();
        self.frames.clear();

        let script_fn = Rc::new(Function {
            name: Some("script".to_string()),
            arity: 0,
            chunk,
        });

        let script_closure = Rc::new(Closure { function: script_fn });

        // calling convention: [callee, arg1..argN]
        self.stack.push(Value::Closure(script_closure));
        self.call_top(0)?;

        while !self.frames.is_empty() {
            let op = {
                let frame = self.frames.last_mut().ok_or("no current frame")?;
                let code = &frame.closure.function.chunk.instructions;

                if frame.ip >= code.len() {
                    return Err("instruction pointer out of bounds".into());
                }

                let op = code[frame.ip];
                frame.ip += 1;
                op
            };

            match op {
                OpCode::OpConstant(i) => {
                    let c = self.current_chunk()?;
                    let v = c
                        .constants
                        .get(i)
                        .cloned()
                        .ok_or_else(|| "constant index out of bounds".to_string())?;
                    self.stack.push(v);
                }

                OpCode::OpAdd => {
                    let b = self.pop()?;
                    let a = self.pop()?;
                    self.stack.push(a.add(b)?);
                }
                OpCode::OpSub => {
                    let b = self.pop_number()?;
                    let a = self.pop_number()?;
                    self.stack.push(Value::Number(a - b));
                }
                OpCode::OpMul => {
                    let b = self.pop_number()?;
                    let a = self.pop_number()?;
                    self.stack.push(Value::Number(a * b));
                }
                OpCode::OpDiv => {
                    let b = self.pop_number()?;
                    let a = self.pop_number()?;
                    self.stack.push(Value::Number(a / b));
                }

                OpCode::OpEq => {
                    let b = self.pop()?;
                    let a = self.pop()?;
                    self.stack.push(Value::Bool(a == b));
                }
                OpCode::OpNotEq => {
                    let b = self.pop()?;
                    let a = self.pop()?;
                    self.stack.push(Value::Bool(a != b));
                }
                OpCode::OpGt => {
                    let b = self.pop_number()?;
                    let a = self.pop_number()?;
                    self.stack.push(Value::Bool(a > b));
                }
                OpCode::OpGe => {
                    let b = self.pop_number()?;
                    let a = self.pop_number()?;
                    self.stack.push(Value::Bool(a >= b));
                }
                OpCode::OpLt => {
                    let b = self.pop_number()?;
                    let a = self.pop_number()?;
                    self.stack.push(Value::Bool(a < b));
                }
                OpCode::OpLe => {
                    let b = self.pop_number()?;
                    let a = self.pop_number()?;
                    self.stack.push(Value::Bool(a <= b));
                }

                OpCode::OpPop => {
                    let _ = self.pop()?;
                }

                OpCode::OpPrint => {
                    let v = self.pop()?;
                    println!("{v}");
                }

                OpCode::OpDefineGlobal(name_const_idx) => {
                    let v = self.pop()?;
                    let c = self.current_chunk()?;
                    let name = c
                        .constants
                        .get(name_const_idx)
                        .ok_or_else(|| "constant index out of bounds".to_string())?
                        .as_str()?
                        .to_string();
                    self.globals.insert(name, v);
                }

                OpCode::OpGetGlobal(name_const_idx) => {
                    let c = self.current_chunk()?;
                    let name = c
                        .constants
                        .get(name_const_idx)
                        .ok_or_else(|| "constant index out of bounds".to_string())?
                        .as_str()?
                        .to_string();
                    let v = self
                        .globals
                        .get(&name)
                        .cloned()
                        .ok_or_else(|| format!("undefined global '{name}'"))?;
                    self.stack.push(v);
                }

                OpCode::OpClosure(fn_const_idx) => {
                    let c = self.current_chunk()?;
                    let v = c
                        .constants
                        .get(fn_const_idx)
                        .cloned()
                        .ok_or_else(|| "constant index out of bounds".to_string())?;
                    let fun_rc = match v {
                        Value::Function(f) => f,
                        _ => return Err("OpClosure expects a Function constant".into()),
                    };
                    let closure_rc = Rc::new(Closure { function: fun_rc });
                    self.stack.push(Value::Closure(closure_rc));
                }

                OpCode::OpCall(argc) => self.call_top(argc)?,

                OpCode::OpJump(target) => {
                    let frame = self.frames.last_mut().ok_or("no current frame")?;
                    frame.ip = target;
                }

                OpCode::OpJumpIfFalse(target) => {
                    let cond = self
                        .stack
                        .last()
                        .ok_or_else(|| "stack underflow (jump cond)".to_string())?
                        .clone();
                    if !Self::is_truthy(&cond) {
                        let frame = self.frames.last_mut().ok_or("no current frame")?;
                        frame.ip = target;
                    }
                }

                OpCode::OpReturn => self.do_return()?,
            }
        }

        Ok(())
    }

    fn is_truthy(v: &Value) -> bool {
        match v {
            Value::Bool(b) => *b,
            Value::Nil => false,
            _ => true,
        }
    }

    fn pop(&mut self) -> Result<Value, String> {
        self.stack.pop().ok_or_else(|| "stack underflow".to_string())
    }

    fn pop_number(&mut self) -> Result<f64, String> {
        match self.pop()? {
            Value::Number(n) => Ok(n),
            _ => Err("type error: expected number".into()),
        }
    }

    fn current_chunk(&self) -> Result<&Chunk, String> {
        let frame = self.frames.last().ok_or_else(|| "no current frame".to_string())?;
        Ok(&frame.closure.function.chunk)
    }

    fn call_top(&mut self, argc: usize) -> Result<(), String> {
        let callee_index = self
            .stack
            .len()
            .checked_sub(argc + 1)
            .ok_or_else(|| "stack underflow in call".to_string())?;

        let callee = self.stack[callee_index].clone();
        let closure_rc: Rc<Closure> = match callee {
            Value::Closure(c) => c,
            Value::Function(f) => Rc::new(Closure { function: f }),
            _ => return Err("can only call function/closure".into()),
        };

        if argc != closure_rc.function.arity {
            return Err(format!(
                "arity mismatch: expected {}, got {}",
                closure_rc.function.arity, argc
            ));
        }

        self.frames.push(CallFrame {
            closure: closure_rc,
            ip: 0,
            stack_base: callee_index,
        });

        Ok(())
    }

    fn do_return(&mut self) -> Result<(), String> {
        let ret = self.pop()?;
        let frame = self.frames.pop().ok_or_else(|| "return with no frame".to_string())?;

        self.stack.truncate(frame.stack_base);
        self.stack.push(ret);

        Ok(())
    }
}
