// sapphire-vm/src/vm.rs
use crate::chunk::{Chunk, OpCode};
//use crate::value::Value;
use crate::Value;

use std::collections::HashMap;

pub struct VM {
    stack: Vec<Value>,
    globals: HashMap<String, Value>,
    ip: usize,
}

impl VM {
    pub fn new() -> Self {
        Self { stack: vec![], globals: HashMap::new(), ip: 0 }
    }

    pub fn run(&mut self, chunk: &Chunk) -> Result<(), String> {
        self.ip = 0;
        while self.ip < chunk.instructions.len() {
            let op = chunk.instructions[self.ip];
            self.ip += 1;

            match op {
                OpCode::OpConstant(i) => self.stack.push(chunk.constants[i].clone()),
                OpCode::OpAdd => {
                    let b = self.stack.pop().ok_or("stack underflow")?;
                    let a = self.stack.pop().ok_or("stack underflow")?;
                    self.stack.push(a.add(b)?); // implement Value::add for type-checking
                }
                OpCode::OpPrint => {
                    let v = self.stack.pop().ok_or("stack underflow")?;
                    println!("{v}");
                }
                OpCode::OpDefineGlobal(name_const_idx) => {
                    let v = self.stack.pop().ok_or("stack underflow")?;
                    let name = chunk.constants[name_const_idx].as_str()?.to_string();
                    self.globals.insert(name, v);
                }
                OpCode::OpGetGlobal(name_const_idx) => {
                    let name = chunk.constants[name_const_idx].as_str()?.to_string();
                    let v = self.globals.get(&name).cloned().ok_or("undefined global")?;
                    self.stack.push(v);
                }
            }
        }
        Ok(())
    }
}
