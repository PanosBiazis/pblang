use sapphire_core::parse;
use sapphire_vm::VM;
use std::io::{self, Write};

fn clear_terminal() {
    // ANSI: clear visible screen + cursor home + (optional) clear scrollback
    print!("\x1B[2J\x1B[H\x1B[3J");

    // Now print an unmistakable marker that we're inside Sapphire
    println!("Sapphire REPL");
    println!();

    let _ = io::stdout().flush();

    // Optional fallback for older Windows consoles (can keep or remove).
    // NOTE: If this runs AFTER the banner, cmd's cls will wipe it again.
    // So either remove this fallback, or run it BEFORE printing the banner.
    #[cfg(windows)]
    {
        // If you still want it, do it first:
        // let _ = std::process::Command::new("cmd").args(["/C", "cls"]).status();
    }
}

fn show_help() {
    println!("Sapphire REPL commands:");
    println!("  help | /h | ?              Show this help");
    println!("  clear | cls | clear screen  Clear the screen");
    println!("  exit | quit                Exit the REPL");
    println!();
    println!("Language:");
    println!("  print <expr>");
    println!("  printf <precision> <expr>");
    println!("  let x=<expr>;   x=<expr>;");
    println!("  if <expr> (<stmts...>)");
    println!("  while <expr> (<stmts...>)");
    println!("  for <name> <count> (<stmts...>)");
}

fn main() {
    let mut vm = VM::new();

    println!("Welcome to Sapphire REPL!");
    loop {
        print!("> ");
        io::stdout().flush().unwrap(); // print! doesn't auto-flush [web:79][web:82]

        let mut input = String::new();
        if io::stdin().read_line(&mut input).is_err() {
            break;
        }

        let raw = input.trim();
        if raw.is_empty() {
            continue;
        }

        // Meta-command prefixes to avoid confusion with language identifiers
        let cmd = raw.strip_prefix(':').unwrap_or(raw);
        let cmd = cmd.strip_prefix('/').unwrap_or(cmd);

        // REPL commands (case-insensitive)
        if cmd.eq_ignore_ascii_case("exit") || cmd.eq_ignore_ascii_case("quit") {
            break;
        }
        if cmd.eq_ignore_ascii_case("help") || cmd.eq_ignore_ascii_case("h") || cmd == "?" {
            show_help();
            continue;
        }
        if cmd.eq_ignore_ascii_case("clear")
            || cmd.eq_ignore_ascii_case("cls")
            || cmd.eq_ignore_ascii_case("clear screen")
        {
            clear_terminal();
            continue;
        }

        // Otherwise treat it as Sapphire code
        match parse(raw) {
            Ok(stmts) => vm.run(stmts),
            Err(e) => println!("Parse error: {:?}", e),
        }
    }
}
