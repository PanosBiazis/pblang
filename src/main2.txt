fn scan_line_for_braces(
    line: &str,
    in_string: &mut bool,
    block_comment_depth: &mut i32,
    brace_depth: &mut i32,
) {
    let mut chars = line.chars().peekable();
    let mut prev_was_backslash = false;

    while let Some(ch) = chars.next() {
        let next = chars.peek().copied();

        // If we're in a string, only care about \" and closing "
        if *in_string {
            if ch == '"' && !prev_was_backslash {
                *in_string = false;
            }
            prev_was_backslash = ch == '\\' && !prev_was_backslash;
            continue;
        }

        // If we're in a block comment, look for nested /* and closing */
        if *block_comment_depth > 0 {
            if ch == '/' && next == Some('*') {
                chars.next(); // consume '*'
                *block_comment_depth += 1;
                continue;
            }
            if ch == '*' && next == Some('/') {
                chars.next(); // consume '/'
                *block_comment_depth -= 1;
                continue;
            }
            continue;
        }

        // Not in string/comment: detect start of line comment //
        if ch == '/' && next == Some('/') {
            break; // ignore rest of line
        }

        // Not in string/comment: detect start of block comment /*
        if ch == '/' && next == Some('*') {
            chars.next(); // consume '*'
            *block_comment_depth += 1;
            continue;
        }

        // Not in string/comment: entering string
        if ch == '"' {
            *in_string = true;
            prev_was_backslash = false;
            continue;
        }

        // Count braces only in normal code
        match ch {
            '{' => *brace_depth += 1,
            '}' => *brace_depth -= 1,
            _ => {}
        }
    }
}

fn strip_comments_mask(src: &str) -> String {
    let bytes = src.as_bytes();
    let mut out = bytes.to_vec(); // same length as input (byte offsets preserved)
    let mut i = 0usize;
    let mut in_string = false;
    let mut block_depth: i32 = 0;

    while i < out.len() {
        let b = out[i];

        // Inside string: only care about escapes and closing "
        if in_string {
            if b == b'\\' {
                i = i.saturating_add(2);
                continue;
            }
            if b == b'"' {
                in_string = false;
            }
            i += 1;
            continue;
        }

        // Inside block comment (nested): mask everything except newlines
        if block_depth > 0 {
            if i + 1 < out.len() && out[i] == b'/' && out[i + 1] == b'*' {
                out[i] = b' ';
                out[i + 1] = b' ';
                block_depth += 1;
                i += 2;
                continue;
            }
            if i + 1 < out.len() && out[i] == b'*' && out[i + 1] == b'/' {
                out[i] = b' ';
                out[i + 1] = b' ';
                block_depth -= 1;
                i += 2;
                continue;
            }
            if out[i] != b'\n' {
                out[i] = b' ';
            }
            i += 1;
            continue;
        }

        // Not in string/comment: entering string
        if b == b'"' {
            in_string = true;
            i += 1;
            continue;
        }

        // line comment //
        if i + 1 < out.len() && out[i] == b'/' && out[i + 1] == b'/' {
            out[i] = b' ';
            out[i + 1] = b' ';
            i += 2;
            while i < out.len() && out[i] != b'\n' {
                out[i] = b' ';
                i += 1;
            }
            continue;
        }

        // block comment /* ... */ (nested)
        if i + 1 < out.len() && out[i] == b'/' && out[i + 1] == b'*' {
            out[i] = b' ';
            out[i + 1] = b' ';
            block_depth = 1;
            i += 2;
            continue;
        }

        i += 1;
    }

    String::from_utf8(out).unwrap()
}

fn line_col(src: &str, byte_idx: usize) -> (usize, usize) {
    let mut line = 1usize;
    let mut col = 1usize;
    for (i, ch) in src.char_indices() {
        if i >= byte_idx {
            break;
        }
        if ch == '\n' {
            line += 1;
            col = 1;
        } else {
            col += 1;
        }
    }
    (line, col)
}

fn get_line(src: &str, line_no: usize) -> &str {
    src.lines().nth(line_no.saturating_sub(1)).unwrap_or("")
}

fn underline_span(src: &str, sp: std::ops::Range<usize>) -> (usize, usize, String, String) {
    let start = sp.start.min(src.len());
    let end = sp.end.min(src.len()).max(start.saturating_add(1));

    let (ln1, col1) = line_col(src, start);
    let (ln2, col2) = line_col(src, end);

    let line = get_line(src, ln1).to_string();

    // If the span is on one line, draw ^^^ under the range.
    // If it spans multiple lines, fall back to a single caret at the start.
    let len = if ln1 == ln2 {
        col2.saturating_sub(col1).max(1)
    } else {
        1
    };

    let marker = format!("{}{}", " ".repeat(col1.saturating_sub(1)), "^".repeat(len));

    (ln1, col1, line, marker)
}

fn format_parse_error(err: &Simple<char>) -> String {
    // Keep this robust across chumsky versions: prefer structured info when available,
    // but fall back to Debug if needed.
    let label = err.label();
    let found = err.found();

    let mut expected = err.expected().copied().flatten().collect::<Vec<char>>();
    expected.sort();
    expected.dedup();

    let base = match err.reason() {
        SimpleReason::Unclosed { span: _, delimiter } => {
            format!("unclosed delimiter '{delimiter}'")
        }
        SimpleReason::Custom(msg) => msg.to_string(),
        SimpleReason::Unexpected => match (found, expected.is_empty()) {
            (Some(f), false) => format!("unexpected '{f}', expected one of: {:?}", expected),
            (Some(f), true) => format!("unexpected '{f}'"),
            (None, false) => format!("unexpected end of input, expected one of: {:?}", expected),
            (None, true) => "parse error".to_string(),
        },
    };

    if let Some(l) = label {
        format!("{base} (while parsing {l})")
    } else {
        base
    }
}

/// Normalize spacing around binary operators (outside string literals). Ensures one space
/// before and after + - * / == != < > <= >=. Does not change content inside "...".
fn fmt_normalize_operator_spacing(line: &str) -> String {
    let mut in_string = false;
    let mut escape_next = false;
    let mut i = 0;
    let bytes = line.as_bytes();
    let mut out = String::with_capacity(line.len() + 16);
    while i < bytes.len() {
        let c = bytes[i] as char;
        if in_string {
            if escape_next {
                escape_next = false;
                out.push(c);
            } else if c == '\\' {
                escape_next = true;
                out.push(c);
            } else if c == '"' {
                in_string = false;
                out.push(c);
            } else {
                out.push(c);
            }
            i += 1;
            continue;
        }
        if c == '"' {
            in_string = true;
            out.push(c);
            i += 1;
            continue;
        }
        let need_space_before = !out.is_empty()
            && !out.ends_with(' ')
            && !out.ends_with('(')
            && !out.ends_with('[')
            && !out.ends_with('=')
            && !out.ends_with('!')
            && !out.ends_with('<')
            && !out.ends_with('>')
            && !out.ends_with(',')
            && !out.ends_with(';');
        // Multi-char operators (longest first)
        let (op_len, need_space_after) = if i + 2 <= bytes.len() {
            let two = std::str::from_utf8(&bytes[i..i + 2]).unwrap_or("");
            if two == "==" || two == "!=" || two == "<=" || two == ">=" {
                (2, true)
            } else {
                (0, false)
            }
        } else {
            (0, false)
        };
        let (op_len, need_space_after) = if op_len > 0 {
            (op_len, need_space_after)
        } else if matches!(c, '+' | '*' | '/' | '<' | '>') {
            (1, true)
        } else if c == '-' {
            // Unary minus (-1): no space after if next is digit and we didn't add space before
            let next_is_digit = i + 1 < bytes.len()
                && matches!(bytes[i + 1] as char, '0'..='9' | '.');
            let space_after = !next_is_digit || need_space_before;
            (1, space_after)
        } else {
            out.push(c);
            i += 1;
            continue;
        };
        let prev_ok = out.ends_with(' ') || out.is_empty()
            || out.ends_with('(')
            || out.ends_with('[')
            || out.ends_with(',')
            || out.ends_with(';');
        if need_space_before && !prev_ok {
            out.push(' ');
        }
        for j in 0..op_len {
            out.push(bytes[i + j] as char);
        }
        i += op_len;
        if need_space_after && i < bytes.len() {
            let next = bytes[i] as char;
            if next != ' ' && next != ')' && next != ']' && next != ';' && next != ',' && next != '\n' {
                out.push(' ');
            }
        }
    }
    out
}

fn print_parse_errors(src: &str, errors: Vec<Simple<char>>) {
    let max = 8usize;
    for (i, err) in errors.iter().take(max).enumerate() {
        let sp = err.span();
        let (ln, col, line, marker) = underline_span(src, sp);

        eprintln!("Parse error at {ln}:{col}: {}", format_parse_error(err));
        eprintln!("{line}");
        eprintln!("{marker}");

        if i + 1 < errors.len().min(max) {
            eprintln!();
        }
    }

    if errors.len() > max {
        eprintln!("(and {} more parse errors)", errors.len() - max);
    }
}

const DEFAULT_MANIFEST: &str = r#"[package]
name = "my-app"
version = "0.1.0"

[dependencies]
"#;

fn copy_dir_all(src: &std::path::Path, dst: &std::path::Path) -> std::io::Result<()> {
    std::fs::create_dir_all(dst)?;
    for e in std::fs::read_dir(src)? {
        let e = e?;
        let ty = e.file_type()?;
        let p = e.path();
        let name = p.file_name().unwrap_or(std::ffi::OsStr::new(""));
        let dest = dst.join(name);
        if ty.is_dir() {
            copy_dir_all(&p, &dest)?;
        } else {
            std::fs::copy(&p, &dest)?;
        }
    }
    Ok(())
}

fn main() {
    let args: Vec<String> = std::env::args().skip(1).collect();
    if args.get(0).map(|s| s.as_str()) == Some("init") {
        let cwd = std::env::current_dir().unwrap_or_else(|_| std::path::PathBuf::from("."));
        let manifest_path = cwd.join("sapphire.toml");
        if manifest_path.exists() {
            eprintln!("sapphire init: {} already exists", manifest_path.display());
            std::process::exit(1);
        }
        match std::fs::write(&manifest_path, DEFAULT_MANIFEST) {
            Ok(()) => println!("created {}", manifest_path.display()),
            Err(e) => {
                eprintln!("sapphire init: failed to write {}: {}", manifest_path.display(), e);
                std::process::exit(1);
            }
        }
        return;
    }
    if args.get(0).map(|s| s.as_str()) == Some("--lock") {
        let start = args
            .get(1)
            .map(|s| std::path::PathBuf::from(s))
            .unwrap_or_else(|| std::env::current_dir().unwrap_or_else(|_| ".".into()));
        let man_dir = sapphire_pm::find_manifest_dir(&start)
            .unwrap_or_else(|| panic!("could not find sapphire.toml from '{}'", start.display()));
        let lock = sapphire_pm::generate_lockfile(&man_dir).expect("failed to generate lockfile");
        let lock_str = sapphire_pm::write_lockfile(&lock).expect("failed to serialize lockfile");
        let lock_path = man_dir.join("sapphire.lock");
        std::fs::write(&lock_path, lock_str).expect("failed to write sapphire.lock");
        println!("wrote {}", lock_path.display());
        return;
    }
    if args.get(0).map(|s| s.as_str()) == Some("install") {
        let rest: Vec<&str> = args.iter().skip(1).map(|s| s.as_str()).collect();
        let mut registry_index_url: Option<&str> = None;
        let mut path_arg: Option<std::path::PathBuf> = None;
        let mut i = 0;
        while i < rest.len() {
            if rest[i] == "--registry-index" {
                i += 1;
                if i < rest.len() {
                    registry_index_url = Some(rest[i]);
                    i += 1;
                } else {
                    eprintln!("sapphire install: --registry-index requires <url>");
                    std::process::exit(1);
                }
            } else {
                path_arg = Some(std::path::PathBuf::from(rest[i]));
                break;
            }
        }
        let start = path_arg
            .unwrap_or_else(|| std::env::current_dir().unwrap_or_else(|_| ".".into()));
        let man_dir = sapphire_pm::find_manifest_dir(&start).unwrap_or_else(|| {
            eprintln!("sapphire install: could not find sapphire.toml from '{}'", start.display());
            std::process::exit(1);
        });
        let require_sha = std::env::var("REQUIRE_REGISTRY_SHA256")
            .map(|v| v == "1" || v.eq_ignore_ascii_case("true"))
            .unwrap_or(false);
        let lock = if let Some(url) = registry_index_url {
            sapphire_pm::generate_lockfile_with_registry(&man_dir, Some(url), require_sha)
        } else {
            sapphire_pm::generate_lockfile(&man_dir)
        };
        let lock = lock.unwrap_or_else(|e| {
            eprintln!("sapphire install: failed to generate lockfile: {}", e);
            std::process::exit(1);
        });
        let lock_str = sapphire_pm::write_lockfile(&lock).unwrap_or_else(|e| {
            eprintln!("sapphire install: failed to serialize lockfile: {}", e);
            std::process::exit(1);
        });
        let lock_path = man_dir.join("sapphire.lock");
        std::fs::write(&lock_path, lock_str).unwrap_or_else(|e| {
            eprintln!("sapphire install: failed to write {}: {}", lock_path.display(), e);
            std::process::exit(1);
        });
        println!("dependencies ready ({})", lock_path.display());
        return;
    }
    if args.get(0).map(|s| s.as_str()) == Some("publish") {
        let rest: Vec<&str> = args.iter().skip(1).map(|s| s.as_str()).collect();
        let mut registry_dir: Option<std::path::PathBuf> = None;
        let mut path_arg: Option<std::path::PathBuf> = None;
        let mut i = 0;
        while i < rest.len() {
            if rest[i] == "--registry-dir" {
                i += 1;
                if i < rest.len() {
                    registry_dir = Some(std::path::PathBuf::from(rest[i]));
                    i += 1;
                } else {
                    eprintln!("sapphire publish: --registry-dir requires <dir>");
                    std::process::exit(1);
                }
            } else {
                path_arg = Some(std::path::PathBuf::from(rest[i]));
                break;
            }
        }
        let registry_dir = registry_dir.unwrap_or_else(|| {
            eprintln!("sapphire publish: --registry-dir <dir> required");
            std::process::exit(1);
        });
        let start = path_arg
            .unwrap_or_else(|| std::env::current_dir().unwrap_or_else(|_| ".".into()));
        let man_dir = sapphire_pm::find_manifest_dir(&start).unwrap_or_else(|| {
            eprintln!("sapphire publish: could not find sapphire.toml from '{}'", start.display());
            std::process::exit(1);
        });
        if let Err(e) = sapphire_pm::publish_to_registry_dir(&man_dir, &registry_dir) {
            eprintln!("sapphire publish: {}", e);
            std::process::exit(1);
        }
        println!("published to {}", registry_dir.display());
        return;
    }
    if args.get(0).map(|s| s.as_str()) == Some("pack") {
        // Stage A executables: create a single .tar.gz artifact with manifest, lockfile and local deps.
        let start = args
            .get(1)
            .map(|s| std::path::PathBuf::from(s))
            .unwrap_or_else(|| std::env::current_dir().unwrap_or_else(|_| ".".into()));
        let man_dir = sapphire_pm::find_manifest_dir(&start).unwrap_or_else(|| {
            eprintln!("sapphire pack: could not find sapphire.toml from '{}'", start.display());
            std::process::exit(1);
        });
        // Ensure we have an up-to-date lockfile before packing.
        let lock = sapphire_pm::generate_lockfile(&man_dir).unwrap_or_else(|e| {
            eprintln!("sapphire pack: failed to generate lockfile: {}", e);
            std::process::exit(1);
        });
        let lock_str = sapphire_pm::write_lockfile(&lock).unwrap_or_else(|e| {
            eprintln!("sapphire pack: failed to serialize lockfile: {}", e);
            std::process::exit(1);
        });
        let lock_path = man_dir.join("sapphire.lock");
        std::fs::write(&lock_path, lock_str).unwrap_or_else(|e| {
            eprintln!("sapphire pack: failed to write {}: {}", lock_path.display(), e);
            std::process::exit(1);
        });

        let default_out = man_dir.join("dist").join("app.tar.gz");
        let out_path = args
            .get(2)
            .map(|s| std::path::PathBuf::from(s))
            .unwrap_or(default_out);
        if let Some(parent) = out_path.parent() {
            if let Err(e) = std::fs::create_dir_all(parent) {
                eprintln!("sapphire pack: failed to create {}: {}", parent.display(), e);
                std::process::exit(1);
            }
        }

        let bytes = sapphire_pm::create_package_tarball(&man_dir).unwrap_or_else(|e| {
            eprintln!("sapphire pack: failed to build tarball: {}", e);
            std::process::exit(1);
        });
        if let Err(e) = std::fs::write(&out_path, &bytes) {
            eprintln!(
                "sapphire pack: failed to write {}: {}",
                out_path.display(),
                e
            );
            std::process::exit(1);
        }
        println!("packed to {}", out_path.display());
        return;
    }
    if args.get(0).map(|s| s.as_str()) == Some("--dump-bytecode") {
        let path = match args.get(1) {
            Some(p) => p,
            None => {
                eprintln!("sapphire --dump-bytecode: pass a .sap or .sapp file");
                std::process::exit(1);
            }
        };
        let cwd = std::env::current_dir().unwrap_or_else(|_| std::path::PathBuf::from("."));
        let file_path = cwd.join(path);
        let src = std::fs::read_to_string(&file_path).unwrap_or_else(|e| {
            eprintln!("sapphire --dump-bytecode: failed to read {}: {}", file_path.display(), e);
            std::process::exit(1);
        });
        let src_for_parse = strip_comments_mask(&src);
        let stmts = match parse(&src_for_parse) {
            Ok(s) => s,
            Err(errs) => {
                eprintln!("{}:", path);
                print_parse_errors(&src_for_parse, errs);
                std::process::exit(2);
            }
        };
        let chunk = compile(stmts);
        println!("{}", disasm(&chunk));
        return;
    }
    if args.get(0).map(|s| s.as_str()) == Some("build") {
        let start = args
            .get(1)
            .map(|s| std::path::PathBuf::from(s))
            .unwrap_or_else(|| std::env::current_dir().unwrap_or_else(|_| ".".into()));
        let man_dir = sapphire_pm::find_manifest_dir(&start)
            .unwrap_or_else(|| {
                eprintln!("sapphire build: could not find sapphire.toml from '{}'", start.display());
                std::process::exit(1);
            });
        let lock = sapphire_pm::generate_lockfile(&man_dir).unwrap_or_else(|e| {
            eprintln!("sapphire build: failed to generate lockfile: {}", e);
            std::process::exit(1);
        });
        let lock_str = sapphire_pm::write_lockfile(&lock).unwrap_or_else(|e| {
            eprintln!("sapphire build: failed to serialize lockfile: {}", e);
            std::process::exit(1);
        });
        let lock_path = man_dir.join("sapphire.lock");
        std::fs::write(&lock_path, lock_str).unwrap_or_else(|e| {
            eprintln!("sapphire build: failed to write {}: {}", lock_path.display(), e);
            std::process::exit(1);
        });
        let entry = man_dir.join("main.sap").canonicalize().ok()
            .or_else(|| man_dir.join("main.sapp").canonicalize().ok())
            .or_else(|| man_dir.join("src").join("main.sap").canonicalize().ok())
            .or_else(|| man_dir.join("src").join("main.sapp").canonicalize().ok());
        if let Some(entry_path) = entry {
            let path_str = entry_path.to_string_lossy();
            let src = match std::fs::read_to_string(&entry_path) {
                Ok(s) => s,
                Err(e) => {
                    eprintln!("sapphire build: failed to read {}: {}", path_str, e);
                    std::process::exit(1);
                }
            };
            match parse(&src) {
                Ok(stmts) => {
                    let diags = type_check(&stmts);
                    if !diags.is_empty() {
                        for d in diags {
                            if let Some(sp) = d.span {
                                eprintln!("{}:{}..{}: {}", path_str, sp.start, sp.end, d.message);
                            } else {
                                eprintln!("{}: {}", path_str, d.message);
                            }
                        }
                        std::process::exit(1);
                    }
                }
                Err(errs) => {
                    eprintln!("{}:", path_str);
                    print_parse_errors(&src, errs);
                    std::process::exit(2);
                }
            }
        }
        println!("build ok");
        return;
    }
    if args.get(0).map(|s| s.as_str()) == Some("bundle") {
        let start = args
            .get(1)
            .map(|s| std::path::PathBuf::from(s))
            .unwrap_or_else(|| std::env::current_dir().unwrap_or_else(|_| ".".into()));
        let man_dir = sapphire_pm::find_manifest_dir(&start).unwrap_or_else(|| {
            eprintln!("sapphire bundle: could not find sapphire.toml from '{}'", start.display());
            std::process::exit(1);
        });
        let out_dir = args
            .get(2)
            .map(|s| std::path::PathBuf::from(s))
            .unwrap_or_else(|| man_dir.join("dist"));
        if let Err(e) = std::fs::create_dir_all(&out_dir) {
            eprintln!("sapphire bundle: failed to create {}: {}", out_dir.display(), e);
            std::process::exit(1);
        }
        let exe_name = if cfg!(target_os = "windows") { "sapphire.exe" } else { "sapphire" };
        let dest_exe = out_dir.join(exe_name);
        if let Ok(current_exe) = std::env::current_exe() {
            if let Err(e) = std::fs::copy(&current_exe, &dest_exe) {
                eprintln!("sapphire bundle: failed to copy executable to {}: {}", dest_exe.display(), e);
                std::process::exit(1);
            }
        }
        for f in &["sapphire.toml", "sapphire.lock"] {
            let src = man_dir.join(f);
            if src.is_file() {
                let _ = std::fs::copy(&src, out_dir.join(f));
            }
        }
        let entry = man_dir.join("main.sap").canonicalize().ok()
            .or_else(|| man_dir.join("main.sapp").canonicalize().ok())
            .or_else(|| man_dir.join("src").join("main.sap").canonicalize().ok())
            .or_else(|| man_dir.join("src").join("main.sapp").canonicalize().ok());
        if let Some(entry_path) = entry {
            let name = entry_path.file_name().unwrap_or(std::ffi::OsStr::new("main.sapp"));
            if let Err(e) = std::fs::copy(&entry_path, out_dir.join(name)) {
                eprintln!("sapphire bundle: failed to copy entry {}: {}", entry_path.display(), e);
                std::process::exit(1);
            }
        }
        let man_bytes = match std::fs::read(man_dir.join("sapphire.toml")) {
            Ok(b) => b,
            Err(_) => {
                println!("bundled to {}", out_dir.display());
                return;
            }
        };
        let man_str = match std::str::from_utf8(&man_bytes) {
            Ok(s) => s,
            Err(_) => {
                println!("bundled to {}", out_dir.display());
                return;
            }
        };
        let man = match sapphire_pm::parse_manifest(man_str) {
            Ok(m) => m,
            Err(_) => {
                println!("bundled to {}", out_dir.display());
                return;
            }
        };
        for (name, spec) in &man.dependencies {
            let rel = spec.path();
            let src = man_dir.join(rel);
            if src.is_dir() {
                let dest = out_dir.join(rel);
                if let Err(e) = copy_dir_all(&src, &dest) {
                    eprintln!("sapphire bundle: failed to copy dep {}: {}", name, e);
                    std::process::exit(1);
                }
            }
        }
        println!("bundled to {}", out_dir.display());
        return;
    }
    if args.get(0).map(|s| s.as_str()) == Some("--check") {
        let path = args.get(1).expect("usage: sapphire --check <file>");
        let src = std::fs::read_to_string(path).expect("failed to read file");
        match parse(&src) {
            Ok(stmts) => {
                let diags = type_check(&stmts);
                if diags.is_empty() {
                    println!("typecheck ok");
                    return;
                }
                for d in diags {
                    if let Some(sp) = d.span {
                        eprintln!("{path}:{}..{}: {}", sp.start, sp.end, d.message);
                    } else {
                        eprintln!("{path}: {}", d.message);
                    }
                }
                std::process::exit(1);
            }
            Err(errs) => {
                print_parse_errors(&src, errs);
                std::process::exit(2);
            }
        }
    }
    if args.get(0).map(|s| s.as_str()) == Some("--doc") {
        let path = args.get(1).expect("usage: sapphire --doc <file> [--html] [out_dir]");
        let html_mode = args.get(2).map(|s| s.as_str()) == Some("--html");
        let out_dir = if html_mode {
            args.get(3).map(String::as_str).unwrap_or("docs")
        } else {
            ""
        };
        let src = std::fs::read_to_string(path).expect("failed to read file");

        fn render_ty(t: &TypeExpr) -> String {
            match &t.kind {
                TypeKind::Any => "any".into(),
                TypeKind::Number => "number".into(),
                TypeKind::Bool => "bool".into(),
                TypeKind::String => "string".into(),
                TypeKind::Nil => "nil".into(),
                TypeKind::Array(inner) => format!("array<{}>", render_ty(inner)),
                TypeKind::Map(inner) => format!("map<{}>", render_ty(inner)),
            }
        }

        match parse_with_polyglot(&src) {
            Ok((stmts, blocks)) => {
                let mut exports: Vec<String> = vec![];
                let mut fns: Vec<String> = vec![];

                for s in &stmts {
                    match &s.kind {
                        StmtKind::Export(names) => {
                            exports.extend(names.iter().cloned());
                        }
                        StmtKind::Fn(name, params, ret, _body) => {
                            let ps = params
                                .iter()
                                .map(|(n, t)| match t {
                                    Some(t) => format!("{n}: {}", render_ty(t)),
                                    None => n.clone(),
                                })
                                .collect::<Vec<_>>()
                                .join(", ");
                            let r = ret.as_ref().map(render_ty);
                            let sig = if let Some(r) = r {
                                format!("fn {name}({ps}) -> {r}")
                            } else {
                                format!("fn {name}({ps})")
                            };
                            fns.push(sig);
                        }
                        _ => {}
                    }
                }

                exports.sort();
                exports.dedup();
                fns.sort();

                if html_mode {
                    let mut html = String::from(
                        r#"<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="utf-8">
  <title>Sapphire module docs</title>
  <style>
    body { font-family: system-ui, sans-serif; max-width: 800px; margin: 1rem auto; padding: 0 1rem; }
    h1 { font-size: 1.5rem; }
    h2 { font-size: 1.2rem; margin-top: 1.5rem; }
    code { background: #f0f0f0; padding: 0.1em 0.3em; border-radius: 3px; }
    ul { list-style: none; padding-left: 0; }
    li { margin: 0.3em 0; }
  </style>
</head>
<body>
  <h1>Module docs</h1>
  <p><strong>file</strong>: <code>"#,
                    );
                    html.push_str(path);
                    html.push_str(r#"</code></p>
  <h2>Exports</h2>
  <ul>
"#);
                    if exports.is_empty() {
                        html.push_str("    <li>(none)</li>\n");
                    } else {
                        for e in &exports {
                            html.push_str("    <li><code>");
                            html.push_str(&e.replace('<', "&lt;").replace('>', "&gt;"));
                            html.push_str("</code></li>\n");
                        }
                    }
                    html.push_str("  </ul>\n  <h2>Top-level functions</h2>\n  <ul>\n");
                    if fns.is_empty() {
                        html.push_str("    <li>(none)</li>\n");
                    } else {
                        for f in &fns {
                            html.push_str("    <li><code>");
                            html.push_str(&f.replace('<', "&lt;").replace('>', "&gt;"));
                            html.push_str("</code></li>\n");
                        }
                    }
                    html.push_str("  </ul>\n  <h2>Polyglot blocks</h2>\n  <ul>\n");
                    if blocks.is_empty() {
                        html.push_str("    <li>(none)</li>\n");
                    } else {
                        for b in blocks {
                            let name = b.name.as_deref().unwrap_or("<unnamed>");
                            let block_args = b.args.as_deref().unwrap_or("");
                            html.push_str(&format!(
                                "    <li><code>@{} {} {}</code> (span {}..{})</li>\n",
                                b.lang, name, block_args, b.span.start, b.span.end
                            ));
                        }
                    }
                    html.push_str("  </ul>\n</body>\n</html>\n");

                    let out_path = std::path::Path::new(out_dir);
                    let _ = std::fs::create_dir_all(out_path);
                    let index = out_path.join("index.html");
                    if let Err(e) = std::fs::write(&index, html) {
                        eprintln!("sapphire --doc --html: failed to write {}: {}", index.display(), e);
                        std::process::exit(1);
                    }
                    println!("wrote {}", index.display());
                } else {
                    println!("## Module docs");
                    println!();
                    println!("- **file**: `{path}`");
                    println!();
                    println!("### Exports");
                    if exports.is_empty() {
                        println!();
                        println!("(none)");
                    } else {
                        for e in &exports {
                            println!("- `{e}`");
                        }
                    }
                    println!();
                    println!("### Top-level functions");
                    if fns.is_empty() {
                        println!();
                        println!("(none)");
                    } else {
                        for f in &fns {
                            println!("- `{f}`");
                        }
                    }
                    println!();
                    println!("### Polyglot blocks");
                    if blocks.is_empty() {
                        println!();
                        println!("(none)");
                    } else {
                        for b in blocks {
                            let name = b.name.as_deref().unwrap_or("<unnamed>");
                            let block_args = b.args.as_deref().unwrap_or("");
                            println!(
                                "- `@{} {}` `{}` (span {}..{})",
                                b.lang, name, block_args, b.span.start, b.span.end
                            );
                        }
                    }
                }
                return;
            }
            Err(errs) => {
                print_parse_errors(&src, errs);
                std::process::exit(2);
            }
        }
    }

    // Format .sap/.sapp files: trim, spacing around operators, indent by brace depth (4 spaces), optional line wrap, ensure final newline
    if args.get(0).map(|s| s.as_str()) == Some("fmt") {
        let cwd = std::env::current_dir().unwrap_or_else(|_| std::path::PathBuf::from("."));
        let mut max_line_length: Option<usize> = None;
        let mut files: Vec<std::path::PathBuf> = Vec::new();
        for a in args.iter().skip(1) {
            if let Some(rest) = a.strip_prefix("--max-line-length=") {
                max_line_length = Some(rest.parse().unwrap_or(100));
            } else if a.as_str() != "--max-line-length" {
                files.push(cwd.join(a));
            }
        }
        if files.is_empty() {
            eprintln!("sapphire fmt: pass one or more .sap or .sapp files (optional: --max-line-length=N)");
            std::process::exit(1);
        }
        const INDENT: &str = "    ";
        for path in files {
            let path = path.canonicalize().unwrap_or_else(|_| path.clone());
            if path
                .extension()
                .and_then(|e| e.to_str())
                .map(|e| e == "sap" || e == "sapp")
                .unwrap_or(false)
            {
                match std::fs::read_to_string(&path) {
                    Ok(src) => {
                        let lines: Vec<&str> = src.lines().collect();
                        let mut out = String::new();
                        let mut depth: i32 = 0;
                        for line in &lines {
                            let mut trimmed = line.trim().to_string();
                            trimmed = fmt_normalize_operator_spacing(&trimmed);
                            if trimmed.is_empty() {
                                out.push('\n');
                                continue;
                            }
                            if trimmed.starts_with('}') {
                                depth = depth.saturating_sub(1);
                            }
                            let indent_str: String = (0..depth.max(0) as usize).map(|_| INDENT).collect();
                            let full_line = format!("{}{}", indent_str, trimmed);
                            if let Some(max_len) = max_line_length {
                                let mut line_to_output = full_line;
                                while !line_to_output.is_empty() {
                                    if line_to_output.len() <= max_len {
                                        out.push_str(&line_to_output);
                                        out.push('\n');
                                        break;
                                    }
                                    let break_at = line_to_output[..max_len].rfind(' ').unwrap_or(max_len);
                                    let break_at = if break_at == 0 { max_len } else { break_at };
                                    let rest = line_to_output.split_off(break_at);
                                    out.push_str(line_to_output.trim_end());
                                    out.push('\n');
                                    line_to_output = format!("{}{}", indent_str, rest.trim_start());
                                }
                            } else {
                                out.push_str(&full_line);
                                out.push('\n');
                            }
                            if trimmed.ends_with('{') {
                                depth += 1;
                            }
                        }
                        if !out.is_empty() && !out.ends_with('\n') {
                            out.push('\n');
                        }
                        if out != src {
                            if let Err(e) = std::fs::write(&path, &out) {
                                eprintln!("sapphire fmt: failed to write {}: {}", path.display(), e);
                                std::process::exit(1);
                            }
                        }
                    }
                    Err(e) => {
                        eprintln!("sapphire fmt: failed to read {}: {}", path.display(), e);
                        std::process::exit(1);
                    }
                }
            }
        }
        return;
    }

    // Lint: parse + typecheck one or more files, report diagnostics
    if args.get(0).map(|s| s.as_str()) == Some("lint") {
        let cwd = std::env::current_dir().unwrap_or_else(|_| std::path::PathBuf::from("."));
        let files: Vec<_> = args.iter().skip(1).map(|s| cwd.join(s)).collect();
        if files.is_empty() {
            eprintln!("sapphire lint: pass one or more .sap or .sapp files");
            std::process::exit(1);
        }
        let mut has_error = false;
        for path in &files {
            let path = path.canonicalize().unwrap_or_else(|_| path.clone());
            let path_str = path.to_string_lossy();
            let src = match std::fs::read_to_string(&path) {
                Ok(s) => s,
                Err(e) => {
                    eprintln!("sapphire lint: {}: {}", path_str, e);
                    has_error = true;
                    continue;
                }
            };
            match parse(&src) {
                Ok(stmts) => {
                    let diags = type_check(&stmts);
                    for d in diags {
                        has_error = true;
                        if let Some(sp) = d.span {
                            eprintln!("{}:{}..{}: {}", path_str, sp.start, sp.end, d.message);
                        } else {
                            eprintln!("{}: {}", path_str, d.message);
                        }
                    }
                }
                Err(errs) => {
                    has_error = true;
                    eprintln!("{}:", path_str);
                    print_parse_errors(&src, errs);
                }
            }
        }
        if has_error {
            std::process::exit(1);
        }
        return;
    }

    // Run test scripts (convention: tests/*.sap or tests/*.sapp)
    if args.get(0).map(|s| s.as_str()) == Some("test") {
        let cwd = std::env::current_dir().unwrap_or_else(|_| std::path::PathBuf::from("."));
        let test_dir = args
            .get(1)
            .map(|s| cwd.join(s))
            .unwrap_or_else(|| cwd.join("tests"));
        let mut passed = 0usize;
        let mut failed = 0usize;
        let mut entries = match std::fs::read_dir(&test_dir) {
            Ok(e) => e,
            Err(e) => {
                eprintln!("sapphire test: cannot read directory '{}': {}", test_dir.display(), e);
                std::process::exit(1);
            }
        }
        .filter_map(|e| e.ok())
        .map(|e| e.path())
        .filter(|p| {
            p.extension()
                .and_then(|e| e.to_str())
                .map(|e| e == "sap" || e == "sapp")
                .unwrap_or(false)
        })
        .collect::<Vec<_>>();
        entries.sort_by(|a, b| a.file_name().cmp(&b.file_name()));

        for path in entries {
            let path_str = path.to_string_lossy().to_string();
            let src = match std::fs::read_to_string(&path) {
                Ok(s) => s,
                Err(e) => {
                    eprintln!("FAIL {}: {}", path_str, e);
                    failed += 1;
                    continue;
                }
            };
            let src_for_parse = strip_comments_mask(&src);
            let stmts = match parse(&src_for_parse) {
                Ok(s) => s,
                Err(errs) => {
                    eprintln!("FAIL {}: parse error", path_str);
                    for err in errs.iter().take(1) {
                        eprintln!("  {}", format_parse_error(err));
                    }
                    failed += 1;
                    continue;
                }
            };
            let mut chunk = compile(stmts);
            chunk.source_name = Some(std::rc::Rc::new(path_str.clone()));
            chunk.source = Some(std::rc::Rc::new(src));
            let mut vm = VM::with_stdlib(StdlibConfig::default());
            match vm.run(chunk) {
                Ok(()) => {
                    passed += 1;
                    println!("ok {}", path_str);
                }
                Err(e) => {
                    failed += 1;
                    eprintln!("FAIL {}: {}", path_str, e.message);
                }
            }
        }

        println!();
        println!("{} passed, {} failed", passed, failed);
        if failed > 0 {
            std::process::exit(1);
        }
        return;
    }

    // Run a script file and exit (no REPL)
    if args.get(0).map(|s| s.as_str()) == Some("run") {
        let path_arg = args.get(1).expect("usage: sapphire run <file.sapp>");
        let cwd = std::env::current_dir().unwrap_or_else(|_| std::path::PathBuf::from("."));
        let script_path = cwd.join(path_arg);
        let script_path = script_path.canonicalize().unwrap_or_else(|_| {
            // When run from target/release, path is relative to repo root
            sapphire_pm::find_manifest_dir(&cwd)
                .and_then(|man| man.join(path_arg).canonicalize().ok())
                .unwrap_or_else(|| cwd.join(path_arg))
        });
        let path_str = script_path.to_string_lossy().to_string();
        let src = match std::fs::read_to_string(&script_path) {
            Ok(s) => s,
            Err(e) => {
                eprintln!("sapphire run: failed to read '{}': {}", script_path.display(), e);
                std::process::exit(1);
            }
        };
        let src_for_parse = strip_comments_mask(&src);
        let stmts = match parse(&src_for_parse) {
            Ok(s) => s,
            Err(errs) => {
                print_parse_errors(&src_for_parse, errs);
                std::process::exit(2);
            }
        };
        let mut chunk = compile(stmts);
        chunk.source_name = Some(std::rc::Rc::new(path_str.clone()));
        chunk.source = Some(std::rc::Rc::new(src.clone()));

        let mut vm = VM::with_stdlib(StdlibConfig::default());
        if let Err(e) = vm.run(chunk) {
            let src_ref = e.source.as_deref().unwrap_or(&src);
            let src_name = e.source_name.as_deref().unwrap_or(&path_str);
            if let Some(sp) = e.span {
                let (ln, col, line, marker) = underline_span(src_ref, sp.start..sp.end);
                eprintln!("Runtime error at {src_name}:{ln}:{col}: {}", e.message);
                eprintln!("{line}");
                eprintln!("{marker}");
            } else {
                eprintln!("Runtime error ({src_name}): {}", e.message);
            }
            if !e.stack.is_empty() {
                eprintln!("Stack trace:");
                for frame in &e.stack {
                    let fname = frame.function.as_deref().unwrap_or("<anonymous>");
                    let fsrc = frame.source_name.as_deref().unwrap_or(&path_str);
                    if let Some(fsp) = frame.span {
                        let (ln, col) = line_col(src_ref, fsp.start);
                        eprintln!("  at {fname} ({fsrc}:{ln}:{col})");
                    } else {
                        eprintln!("  at {fname} ({fsrc})");
                    }
                }
            }
            std::process::exit(1);
        }
        return;
    }

    let mut vm = VM::with_stdlib(StdlibConfig::default());

    println!("Welcome to Sapphire REPL!");
    println!("Type 'help' for commands.");

    let mut buf = String::new();
    let mut brace_depth: i32 = 0;
    let mut in_string = false;
    let mut block_comment_depth: i32 = 0;

    loop {
        if buf.is_empty() {
            print!("> ");
        } else {
            print!("... ");
        }
        if io::stdout().flush().is_err() {
            break;
        }

        let mut line = String::new();
        if io::stdin().read_line(&mut line).is_err() {
            break;
        }

        let trimmed_end = line.trim_end();

        // Commands only when starting fresh (not mid-block).
        if buf.is_empty() {
            let raw = trimmed_end.trim();
            if raw.is_empty() {
                continue;
            }

            let cmd = raw.strip_prefix(':').unwrap_or(raw);
            let cmd = cmd.strip_prefix('/').unwrap_or(cmd);

            if cmd.eq_ignore_ascii_case("exit") || cmd.eq_ignore_ascii_case("quit") {
                break;
            }
            if cmd.eq_ignore_ascii_case("help") || cmd.eq_ignore_ascii_case("h") || cmd == "?" {
                show_help();
                continue;
            }
            if cmd.eq_ignore_ascii_case("clear") || cmd.eq_ignore_ascii_case("cls") {
                clear_terminal();
                continue;
            }
        }

        buf.push_str(trimmed_end);
        buf.push('\n');

        scan_line_for_braces(
            trimmed_end,
            &mut in_string,
            &mut block_comment_depth,
            &mut brace_depth,
        );

        // Only attempt parse when braces balanced AND we're not inside a string.
        if brace_depth <= 0 && !in_string && block_comment_depth == 0 {
            // Keep byte offsets stable: never trim the string we parse/execute.
            let src_for_spans = strip_comments_mask(&buf);
            if !src_for_spans.trim().is_empty() {
                match parse(&src_for_spans) {
                    Ok(stmts) => {
                        let mut chunk = compile(stmts);
                        // Attach source for accurate runtime diagnostics.
                        chunk.source_name = Some(std::rc::Rc::new("<repl>".to_string()));
                        chunk.source = Some(std::rc::Rc::new(buf.clone()));
                        if let Err(e) = vm.run(chunk) {
                            let src = e.source.as_deref().unwrap_or(&buf);
                            let src_name = e.source_name.as_deref().unwrap_or("<repl>");
                            if let Some(sp) = e.span {
                                let (ln, col) = line_col(src, sp.start);
                                eprintln!("Runtime error at {src_name}:{ln}:{col}: {}", e.message);
                                let line = get_line(src, ln);
                                eprintln!("{line}");
                                eprintln!("{}^", " ".repeat(col.saturating_sub(1)));
                            } else {
                                eprintln!("Runtime error ({src_name}): {}", e.message);
                            }

                            if !e.stack.is_empty() {
                                eprintln!("Stack trace:");
                                for frame in &e.stack {
                                    let fname = frame.function.as_deref().unwrap_or("<anonymous>");
                                    let fsrc = frame.source_name.as_deref().unwrap_or("<repl>");
                                    if let Some(fsp) = frame.span {
                                        let (ln, col) = line_col(src, fsp.start);
                                        eprintln!("  at {fname} ({fsrc}:{ln}:{col})");
                                    } else {
                                        eprintln!("  at {fname} ({fsrc})");
                                    }
                                }
                            }
                        }
                    }
                    Err(e) => {
                        print_parse_errors(&buf, e);
                    }
                }
            }

            buf.clear();
            brace_depth = 0;
            in_string = false;
            block_comment_depth = 0;
        }
    }
}
